    Formulas used 
    MASS         m_i = A_i * L_i * 1e-9 * rho_i                      [kg]
REACH        S_i = sum(L_j, j >= i),  a_i = S_i + cg_offset      [mm]
SELF WEIGHT  Sig_i = sum_{j>=i} [ m_j(S_i-S_j+L_j/2) + P_j(S_i-S_j+L_j) ]
DEMAND       M_i = Kd * g * (Payload * a_i + Sig_i)              [N.mm]
CAPACITY     V_i = Z_i * yield_i * k / FoS                       [N.mm]
             W_i = T_rated * qty * derate * ratio * eta * 1000   [N.mm]
             X_i = min(V_i, W_i)
PAYLOAD      P_i = (X_i / (Kd*g) - Sig_i) / a_i,  limit = min_i  [kg]
DEFLECTION   th_i = (F L^2/2 + Me L + w L^3/6) / (E I)           [rad]
             d_i  = (F L^3/3 + Me L^2/2 + w L^4/8) / (E I)       [mm]
             droop = sum(d_i) + sum(th_i * S_{i+1})
DYNAMICS     T_req = g*(Payload*a_i + Sig_i)/1000 + J_i*alpha    [N.m]
FREQUENCY    f_n ~ (1/2pi) * sqrt(3 E I / (m L^3))               [Hz]
